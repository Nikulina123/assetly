# Runtime dependencies -- the single source of truth for them.
#
# This list lives at the repo root, literally, because Vercel's Python builder
# installs from a root requirements.txt and does NOT support pip's `-r` include
# directive: pointing this file at backend/requirements.txt fails the build with
# "could not parse requirements.txt: Error parsing included file". Test-only
# dependencies are in backend/requirements-dev.txt, which pip reads normally and
# which chains back to this file.
# Upgraded to 0.141.1 to pull starlette >= 1.1.0, which fixes PYSEC-2026-161,
# 249, 248, 1943, 1941, 2281, 2280. Older versions lock starlette to 0.38.6 which
# has known advisories.
fastapi==0.141.1
# Pinned explicitly, not left to FastAPI's transitive floor: the CVE fix
# above (starlette >= 1.1.0) currently rides on whatever fastapi==0.141.1
# happens to require, and a future resolver run against a loosened fastapi
# pin could walk starlette back down without anyone noticing. 1.6.0 is the
# version actually installed and audited (see backend/requirements-dev.txt's
# pip-audit).
starlette==1.6.0
uvicorn[standard]==0.30.6
asyncpg==0.29.0
pydantic==2.9.2
python-dotenv==1.2.2
bcrypt==4.2.0
jinja2==3.1.6
# 0.0.18 is the CVE-2024-53981 floor (resource exhaustion parsing malformed
# multipart input). Every admin POST route accepts multipart form data, so
# this is directly reachable. Do not pin below this.
# Updated to 0.0.31 to fix PYSEC-2026-3040 and related vulnerabilities
python-multipart==0.0.31
itsdangerous==2.2.0
# TOTP (RFC 6238) for admin MFA. Hand-rolling the HMAC is ~40 lines, but the
# QR encoder next to it is not, and a subtly wrong provisioning URI is a
# silently broken enrollment that only shows up when someone is locked out.
pyotp==2.9.0
# Pure-Python QR encoder, no Pillow and no native code. Renders the
# otpauth:// URI as an inline <svg> so enrollment needs no CDN fetch and no
# image endpoint.
segno==1.6.1
# Fernet, for encrypting stored TOTP seeds. Was previously dev-only (used by
# scripts/sign_release.py); it is a runtime dependency now that app/mfa.py
# imports it. Ships manylinux wheels, so the Vercel build is unaffected.
#
# 50.0.1, not the 43.0.1 this arrived on: moving this pin from the dev list to
# the runtime list also moved it into pip-audit's --strict scan of THIS file,
# which failed the build on six advisories against 43.0.1 (PYSEC-2026-35,
# -1284, -2141, -3553, -3554 and GHSA-537c-gmf6-5ccf). Their fix versions range
# up to 49.0.0, so anything below that still fails the scan. Pinned to the
# current release rather than to the 49.0.0 floor, for headroom before the next
# advisory forces this same bump again.
#
# The API surface used here is narrow and stable across these majors -- Fernet
# and HKDF in app/mfa.py, serialization plus asymmetric padding in
# scripts/sign_release.py -- and the full suite passes on it.
cryptography==50.0.1
